cmake_minimum_required(VERSION 3.20)
project(desk_host_checks LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(DESK_SANITIZE "Address and undefined behaviour sanitizers" ON)
if(DESK_SANITIZE AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
include(FetchContent)
FetchContent_Declare(googletest URL https://github.com/google/googletest/archive/refs/tags/v1.15.2.tar.gz DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(googletest)
set(ROOT ${CMAKE_CURRENT_LIST_DIR}/..)
add_library(desk_logic STATIC ${ROOT}/telematics/telematics.c ${ROOT}/vehicle_alert/alert.c ${ROOT}/firmware/components/desk_radio/desk_radio.c ${ROOT}/firmware/components/desk_core/desk_core.c ${ROOT}/firmware/tests/vendor/cJSON.c)
target_include_directories(desk_logic PUBLIC ${ROOT}/telematics ${ROOT}/vehicle_alert ${ROOT}/firmware/components/desk_radio/include ${ROOT}/firmware/components/desk_core/include ${ROOT}/firmware/tests/vendor)
if(NOT MSVC)
  target_compile_options(desk_logic PRIVATE -Wall -Wextra -Werror -Wno-deprecated-declarations)
  target_link_libraries(desk_logic PUBLIC m)
endif()
add_executable(telematics-replay ${ROOT}/telematics/replay.c)
target_link_libraries(telematics-replay desk_logic)
add_executable(desk-tests tests.cpp)
target_link_libraries(desk-tests desk_logic GTest::gtest_main)
enable_testing()
include(GoogleTest)
gtest_discover_tests(desk-tests DISCOVERY_TIMEOUT 30)
add_library(replay_parser STATIC ${ROOT}/replay/replay.c)
target_compile_definitions(replay_parser PRIVATE DESK_REPLAY_LIBRARY)
target_include_directories(replay_parser PUBLIC ${ROOT}/replay)
target_link_libraries(replay_parser desk_logic)
target_link_libraries(desk-tests replay_parser)
add_executable(replay-debug ${ROOT}/replay/replay.c)
target_compile_definitions(replay-debug PRIVATE DESK_REPLAY_DEBUG)
target_link_libraries(replay-debug desk_logic)
add_library(stm32_drivers STATIC ${ROOT}/stm32/src/drivers.cpp)
target_include_directories(stm32_drivers PUBLIC ${ROOT}/stm32/include)
target_link_libraries(desk-tests stm32_drivers)
