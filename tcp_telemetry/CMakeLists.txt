cmake_minimum_required(VERSION 3.20)
project(desk_tcp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(DESK_TSAN "ThreadSanitizer" OFF)
if(DESK_TSAN)
 add_compile_options(-fsanitize=thread -fno-omit-frame-pointer -g)
 add_link_options(-fsanitize=thread)
endif()
find_package(Threads REQUIRED)
add_executable(desk-tcp server.cpp)
target_link_libraries(desk-tcp PRIVATE Threads::Threads)
if(WIN32)
 target_link_libraries(desk-tcp PRIVATE ws2_32)
else()
 target_compile_options(desk-tcp PRIVATE -Wall -Wextra -Werror -Wno-misleading-indentation)
endif()
